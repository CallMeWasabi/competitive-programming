2. Monotonic stack / deque

// ตัวแรกทางซ้ายที่เล็กกว่า a[i]
vector<int> L(n), st;
for (int i = 0; i < n; i++) {
    while (!st.empty() && a[st.back()] >= a[i]) st.pop_back();
    L[i] = st.empty() ? -1 : st.back();
    st.push_back(i);
}
// ใช้หา "สี่เหลี่ยมใหญ่สุดใน histogram": แต่ละ i ขยายได้ระหว่าง L[i]+1 ถึง R[i]-1

// min ของทุก window ขนาด k
deque<int> dq;
for (int i = 0; i < n; i++) {
    while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back();
    dq.push_back(i);
    if (dq.front() <= i - k) dq.pop_front();
    if (i >= k - 1) ans.push_back(a[dq.front()]);
}

3. Sparse table (static RMQ)

int LOG = __lg(n) + 1;
vector<vector<int>> sp(LOG, vector<int>(n));
sp[0] = a;
for (int j = 1; j < LOG; j++)
    for (int i = 0; i + (1 << j) <= n; i++)
        sp[j][i] = min(sp[j-1][i], sp[j-1][i + (1 << (j-1))]);
auto query = [&](int l, int r) {          // [l, r] รวมปลาย
    int j = __lg(r - l + 1);
    return min(sp[j][l], sp[j][r - (1 << j) + 1]);
};

4. Fenwick tree (BIT)

struct BIT {
    int n; vector<ll> t;
    BIT(int n) : n(n), t(n + 1, 0) {}
    void add(int i, ll v) { for (i++; i <= n; i += i & -i) t[i] += v; }      // 0-indexed
    ll sum(int i) { ll r = 0; for (i++; i > 0; i -= i & -i) r += t[i]; return r; } // a[0..i]
    ll sum(int l, int r) { return sum(r) - (l ? sum(l - 1) : 0); }
};
// range add + point query: add(l, v); add(r+1, -v); ค่าที่ i = sum(i)
// นับ inversion: วนจากซ้าย, ans += (จำนวนที่ใส่ไปแล้วซึ่ง > a[i]) แล้ว add(a[i], 1) (compress ค่าก่อน)

5. Segment tree (+lazy)

เขียน segment tree แบบ generic ครั้งเดียวใน notebook แล้วเปลี่ยนแค่ Node, merge, และวิธี apply lazy — ทุกโจทย์ "ข้อมูลรวมของช่วงที่ merge ได้" ใช้ได้หมด (max subarray sum ในช่วง = เก็บ {sum, best prefix, best suffix, best})

// Range add, range sum (ปรับ merge/apply ตามโจทย์)
struct Seg {
    int n; vector<ll> t, lz;
    Seg(int n) : n(n), t(4 * n, 0), lz(4 * n, 0) {}
    void apply(int x, int len, ll v) { t[x] += v * len; lz[x] += v; }
    void push(int x, int l, int r) {
        if (!lz[x]) return;
        int m = (l + r) / 2;
        apply(2*x, m - l + 1, lz[x]); apply(2*x+1, r - m, lz[x]); lz[x] = 0;
    }
    void update(int ql, int qr, ll v, int x, int l, int r) {
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { apply(x, r - l + 1, v); return; }
        push(x, l, r);
        int m = (l + r) / 2;
        update(ql, qr, v, 2*x, l, m); update(ql, qr, v, 2*x+1, m+1, r);
        t[x] = t[2*x] + t[2*x+1];
    }
    ll query(int ql, int qr, int x, int l, int r) {
        if (qr < l || r < ql) return 0;
        if (ql <= l && r <= qr) return t[x];
        push(x, l, r);
        int m = (l + r) / 2;
        return query(ql, qr, 2*x, l, m) + query(ql, qr, 2*x+1, m+1, r);
    }
    void update(int l, int r, ll v) { update(l, r, v, 1, 0, n - 1); }
    ll query(int l, int r) { return query(l, r, 1, 0, n - 1); }
};
Walk on segment tree"หาตำแหน่งแรกที่ค่า ≥ x" (Hotel Queries) ทำได้ใน O(log n) โดยเดินลงต้นไม้: ถ้าลูกซ้าย max ≥ x ลงซ้าย ไม่งั้นลงขวา — ไม่ต้อง binary search ครอบ query อีกชั้น (ที่จะเป็น O(log² n))
Lazy สองชนิด (กำหนดค่า + บวก) ต้องคิดลำดับการประกอบ lazy ให้ถูก: "กำหนดค่า" ล้าง "บวก" ที่ค้างอยู่ — ทดสอบด้วย brute force เสมอ

6. Tree basics & diameter

// DFS แบบ iterative: ได้ parent, depth, ลำดับ (pre-order) โดยไม่มีปัญหา stack
vector<int> par(n, -1), dep(n, 0), order; order.reserve(n);
vector<int> st = {root}; par[root] = root;
while (!st.empty()) {
    int v = st.back(); st.pop_back(); order.push_back(v);
    for (int u : adj[v]) if (u != par[v]) { par[u] = v; dep[u] = dep[v] + 1; st.push_back(u); }
}
// ประมวลผลจากใบขึ้นราก = วน order ย้อนหลัง
vector<int> sz(n, 1);
for (int i = n - 1; i > 0; i--) sz[par[order[i]]] += sz[order[i]];

// Diameter: BFS จากจุดใดก็ได้ → จุดไกลสุด a; BFS จาก a → จุดไกลสุด b; dist(a,b) คือ diameter

7. Tree DP & rerooting

Tree DP: dp[v] คำนวณจาก dp ของลูก — เช่น maximum matching บน tree: dp[v][0/1] = คำตอบใน subtree ของ v เมื่อ v ยังว่าง/ถูกจับคู่แล้ว

Rerooting (ถามคำตอบเมื่อแต่ละ node เป็นราก): รอบแรกคำนวณ "ข้อมูลด้านล่าง" จากใบขึ้น, รอบสองส่ง "ข้อมูลด้านบน" จากรากลง

// ผลรวมระยะทางจาก v ไปทุก node (Tree Distances II)
// down[v] = ผลรวมระยะทางไป node ใน subtree ของ v
for (int i = n - 1; i >= 0; i--) { int v = order[i];
    for (int u : adj[v]) if (u != par[v]) down[v] += down[u] + sz[u]; }
// ans[root] = down[root]; ย้ายรากจาก v ไปลูก u: node ใน subtree u ใกล้ขึ้น 1, ที่เหลือไกลขึ้น 1
ans[root] = down[root];
for (int v : order) for (int u : adj[v]) if (u != par[v])
    ans[u] = ans[v] - sz[u] + (n - sz[u]);

8. Euler tour: subtree → ช่วง

// tin[v]..tout[v] คือช่วงของ subtree v ใน array แบน
int timer = 0;
function<void(int,int)> dfs = [&](int v, int p) {
    tin[v] = timer++;
    for (int u : adj[v]) if (u != p) dfs(u, v);
    tout[v] = timer - 1;
};
// subtree sum ของ v = BIT.sum(tin[v], tout[v]) ; เปลี่ยนค่า v = BIT.add(tin[v], delta)
// path root→v sum: บวก val ที่ tin[v], ลบ val ที่ tout[v]+1 แล้ว prefix sum ถึง tin[v]

9. Binary lifting & LCA

int LOG = 18;                               // 2^18 > 2·10^5
vector<vector<int>> up(LOG, vector<int>(n));
up[0] = par;                                // par[root] = root
for (int j = 1; j < LOG; j++) for (int v = 0; v < n; v++) up[j][v] = up[j-1][up[j-1][v]];

auto kth = [&](int v, int k) {              // บรรพบุรุษลำดับที่ k
    for (int j = 0; j < LOG; j++) if (k >> j & 1) v = up[j][v];
    return v;
};
auto lca = [&](int a, int b) {
    if (dep[a] < dep[b]) swap(a, b);
    a = kth(a, dep[a] - dep[b]);
    if (a == b) return a;
    for (int j = LOG - 1; j >= 0; j--) if (up[j][a] != up[j][b]) a = up[j][a], b = up[j][b];
    return up[0][a];
};
// dist(a,b) = dep[a] + dep[b] - 2*dep[lca(a,b)]
// นับ path ผ่านแต่ละ node: cnt[a]++, cnt[b]++, cnt[lca]--, cnt[par[lca]]-- แล้วรวมจากใบขึ้นราก

Binary lifting ใช้ได้กับ functional graph ด้วย (ทุก node มี edge ออกหนึ่งเส้น — "หลังจากกระโดด k ครั้งอยู่ที่ไหน", Planets Queries I)

10. เทคนิค offline ที่ควรรู้จัก

  • Sort query: Distinct Values Queries — เรียง query ตาม r แล้วเก็บ "ตำแหน่งล่าสุดของแต่ละค่า" ใน BIT

  • ภาคต่อ: Heavy-Light Decomposition (path query + update), Mo's algorithm, persistent segment tree
