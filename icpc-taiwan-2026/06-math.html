1. Modular arithmetic

  • บวก ลบ คูณ: mod หลังทุก operation · ลบ: ((a - b) % M + M) % M เพราะ % ของ C++ ให้ค่าลบได้
  • หาร: คูณด้วย inverse — a / b ≡ a · bM−2 (mod M) เมื่อ M เป็นจำนวนเฉพาะ และ b ไม่หารด้วย M
  • คูณ ll สองตัวที่ < 1018 mod ใหญ่: ใช้ (__int128)a * b % M
  • เลขชี้กำลังใหญ่: ae ≡ ae mod (M−1) (mod M) เมื่อ M เป็นจำนวนเฉพาะและ a ไม่หารด้วย M (Fermat) — Exponentiation II
const ll MOD = 1e9 + 7;
ll power(ll b, ll e, ll m = MOD) {
    ll r = 1; b %= m;
    while (e > 0) {
        if (e & 1) r = r * b % m;
        b = b * b % m; e >>= 1;
    }
    return r;
}
ll inv(ll a) { return power(a, MOD - 2); }      // MOD ต้องเป็นจำนวนเฉพาะ

2. GCD & extended Euclid

// std::gcd, std::lcm มีใน C++17 (<numeric>) — lcm อาจ overflow: a / gcd(a,b) * b
// ext_gcd: หา x, y ที่ a·x + b·y = g = gcd(a, b)
ll ext_gcd(ll a, ll b, ll &x, ll &y) {
    if (b == 0) { x = 1; y = 0; return a; }
    ll x1, y1, g = ext_gcd(b, a % b, x1, y1);
    x = y1; y = x1 - (a / b) * y1;
    return g;
}
// inverse ของ a mod m (m ไม่ต้องเป็นจำนวนเฉพาะ แต่ gcd(a,m) ต้อง = 1):
// ext_gcd(a, m, x, y); inv = (x % m + m) % m
// a·x + b·y = c มีคำตอบจำนวนเต็ม ⇔ gcd(a,b) | c

CRT (ระบบ x ≡ ai mod mi): รวมทีละสองสมการด้วย ext_gcd — รายละเอียดและโจทย์อยู่ใน docs/advance_cp_topic.md หัวข้อ 1

3. Primes, sieve, divisors

// Sieve เก็บ smallest prime factor: แยกตัวประกอบเลข ≤ N ได้ใน O(log x)
const int N = 1e6 + 1;
vector<int> spf(N, 0), primes;
for (int i = 2; i < N; i++) {
    if (spf[i] == 0) { spf[i] = i; primes.push_back(i); }
    for (int p : primes) {
        if (p > spf[i] || (ll)i * p >= N) break;
        spf[i * p] = p;                          // linear sieve: แต่ละตัวถูกตั้งค่าครั้งเดียว
    }
}
// แยกตัวประกอบ x ≤ N
while (x > 1) { int p = spf[x]; int c = 0; while (x % p == 0) x /= p, c++; /* p^c */ }

// แยกตัวประกอบ x ≤ 10^12 แบบ trial division O(√x)
for (ll p = 2; p * p <= x; p++) while (x % p == 0) { fac.push_back(p); x /= p; }
if (x > 1) fac.push_back(x);
  • ถ้า n = ∏ piei: จำนวนตัวหาร = ∏(ei+1), ผลรวมตัวหาร = ∏(piei+1−1)/(pi−1)
  • φ(n) = n ∏(1 − 1/p) — จำนวนเลขใน [1, n] ที่ coprime กับ n
  • นับตัวหารของทุกเลข ≤ N: วน d แล้ววนพหุคูณของ d → O(N log N) (harmonic series เป็นเทคนิคที่ใช้บ่อยมาก)
  • ⌊n/i⌋ มีค่าต่างกันแค่ O(√n) ค่า — รวมกลุ่ม i ที่ให้ค่าเดียวกัน: for (l = 1; l <= n; l = r + 1) { r = n / (n / l); ... }

4. Combinatorics

const int MX = 2e6 + 5;
ll fact[MX], ifact[MX];
void init() {
    fact[0] = 1;
    for (int i = 1; i < MX; i++) fact[i] = fact[i-1] * i % MOD;
    ifact[MX-1] = power(fact[MX-1], MOD - 2);
    for (int i = MX - 1; i > 0; i--) ifact[i-1] = ifact[i] * i % MOD;
}
ll C(int n, int k) {
    if (k < 0 || k > n) return 0;
    return fact[n] * ifact[k] % MOD * ifact[n-k] % MOD;
}

5. Inclusion–exclusion

// นับเลขใน [1, n] ที่หารด้วย a_i อย่างน้อยหนึ่งตัว (k ≤ 20)
ll total = 0;
for (int mask = 1; mask < (1 << k); mask++) {
    ll prod = 1; bool over = false;
    for (int i = 0; i < k; i++) if (mask >> i & 1) {
        if (prod > n / a[i]) { over = true; break; }   // กัน overflow
        prod *= a[i];                                   // ถ้า a ไม่ coprime กัน ใช้ lcm
    }
    if (over) continue;
    total += (__builtin_popcount(mask) & 1 ? 1 : -1) * (n / prod);
}

Pattern "นับสิ่งที่ ไม่ มีคุณสมบัติใดเลย" = ทั้งหมด − (inclusion–exclusion ของ "มีคุณสมบัติอย่างน้อยหนึ่ง") ใช้กับ coprime pairs (ร่วมกับ Möbius), นับ surjection, grid paths ที่เลี่ยงกับดัก

6. Matrix exponentiation

Linear recurrence ที่ n ถึง 1018 (Fibonacci, จำนวนวิธีโยนลูกเต๋า, จำนวน path ยาว k ในกราฟ) → ยกกำลัง matrix ขนาด d×d ใน O(d3 log n)

using Mat = vector<vector<ll>>;
Mat mul(const Mat &A, const Mat &B) {
    int n = A.size(), m = B[0].size(), k = B.size();
    Mat C(n, vector<ll>(m, 0));
    for (int i = 0; i < n; i++) for (int t = 0; t < k; t++) if (A[i][t])
        for (int j = 0; j < m; j++) C[i][j] = (C[i][j] + A[i][t] * B[t][j]) % MOD;
    return C;
}
Mat mpow(Mat A, ll e) {
    int n = A.size(); Mat R(n, vector<ll>(n, 0));
    for (int i = 0; i < n; i++) R[i][i] = 1;
    for (; e; e >>= 1, A = mul(A, A)) if (e & 1) R = mul(R, A);
    return R;
}

// dice: f(n) = f(n-1)+...+f(n-6) → matrix 6x6 แถวแรกเป็น 1 ทั้งหมด, ใต้ diagonal เป็น 1

7. Probability & expected value

  • Linearity of expectation: E[X+Y] = E[X]+E[Y] เสมอ แม้ X, Y ไม่อิสระ — แตกปัญหาเป็นผลรวมของ indicator "เหตุการณ์ i เกิดไหม" แล้วรวมความน่าจะเป็น
  • E[max] ของ n ค่า: ใช้ P(max ≤ k) = ∏ P(Xi ≤ k) แล้ว E = Σ k·(P(max ≤ k) − P(max ≤ k−1))
  • DP ความน่าจะเป็น: dp[i][s] = ความน่าจะเป็นที่หลัง i ครั้งได้ผลรวม s
  • คำตอบ "เป็นเศษส่วน mod p": คำนวณ p·q−1 mod M ตลอดทาง
  • พิมพ์ทศนิยม: อ่านกติกาการปัดให้ดี เช่น CSES Candy Lottery / Moving Robots ให้ปัดแบบ round half to even

8. Game theory

  • Win/lose positions: สถานะแพ้ = ทุก move ไปสถานะชนะ, สถานะชนะ = มี move ไปสถานะแพ้ — DP/BFS ย้อนจากสถานะจบ
  • Nim: ผู้เล่นแรกชนะ ⇔ XOR ของทุกกอง ≠ 0
  • Sprague–Grundy: เกมที่เป็นผลรวมของเกมย่อยอิสระ → grundy(สถานะ) = mex ของ grundy สถานะถัดไป; ผลรวมเกม ชนะ ⇔ XOR ของ grundy ≠ 0
  • ถ้าหา pattern ไม่ได้: brute-force grundy ของ n เล็กๆ แล้วพิมพ์ดู — มักเป็นคาบ
