1. กระบวนการออกแบบ DP (ทำบนกระดาษทุกครั้ง)

  1. State: เขียนเป็นประโยค "dp[i][j] = คำตอบ ของ ปัญหาย่อย ที่ ..." ถ้าเขียนประโยคนี้ไม่ได้ชัด อย่าเริ่มพิมพ์
  2. ถามว่า state พอไหม: การตัดสินใจขั้นต่อไปต้องรู้อะไรบ้างจากอดีต? สิ่งที่ต้องรู้ทั้งหมดต้องอยู่ใน state
  3. Transition: "การตัดสินใจสุดท้าย" มีกี่แบบ แต่ละแบบมาจาก state ไหน
  4. Base case และลำดับการคำนวณ: state ที่ถูกอ้างอิงต้องคำนวณก่อน
  5. Complexity: (จำนวน state) × (ค่า transition ต่อ state) ต้องอยู่ใน budget — ถ้าไม่ผ่าน ลองลดมิติหรือหาทางทำ transition เร็วขึ้น
// Top-down (memoization): เขียนง่าย ตรงกับความคิด แต่ recursion ลึกอาจ stack overflow
ll memo[N]; bool vis[N];
ll f(int i) {
    if (i == 0) return 1;
    if (vis[i]) return memo[i];
    vis[i] = true;
    ll r = 0;
    for (int d = 1; d <= 6 && d <= i; d++) r = (r + f(i - d)) % MOD;
    return memo[i] = r;
}

// Bottom-up: เร็วกว่า, ลด memory ได้ (เก็บแค่แถวก่อนหน้า)
vector<ll> dp(n + 1, 0); dp[0] = 1;
for (int i = 1; i <= n; i++)
    for (int d = 1; d <= 6 && d <= i; d++) dp[i] = (dp[i] + dp[i - d]) % MOD;

2. Counting & coin problems

ความต่างเล็กๆ ที่คนพลาดบ่อย: นับแบบสนใจลำดับ vs ไม่สนใจลำดับ เกิดจากลำดับของ loop

// นับลำดับ (1+2 กับ 2+1 นับต่างกัน) — Coin Combinations I
for (int s = 1; s <= X; s++)
    for (int c : coins) if (c <= s) dp[s] = (dp[s] + dp[s - c]) % MOD;

// นับชุด (ไม่สนลำดับ) — Coin Combinations II: เหรียญอยู่ loop นอก
for (int c : coins)
    for (int s = c; s <= X; s++) dp[s] = (dp[s] + dp[s - c]) % MOD;

// จำนวนเหรียญน้อยที่สุด
vector<int> best(X + 1, INF); best[0] = 0;
for (int s = 1; s <= X; s++)
    for (int c : coins) if (c <= s) best[s] = min(best[s], best[s - c] + 1);

3. Knapsack

// 0/1 knapsack: dp[w] = มูลค่ามากที่สุดเมื่อใช้น้ำหนัก <= w
vector<ll> dp(W + 1, 0);
for (int i = 0; i < n; i++)
    for (int w = W; w >= wt[i]; w--)          // วนถอยหลัง: ของแต่ละชิ้นใช้ได้ครั้งเดียว
        dp[w] = max(dp[w], dp[w - wt[i]] + val[i]);
// unbounded (ใช้ซ้ำได้): วน w ไปข้างหน้าแทน

// subset sum ด้วย bitset: เร็วขึ้น ~64 เท่า
bitset<100001> can; can[0] = 1;
for (int x : a) can |= can << x;
เมื่อ W ใหญ่แต่ value เล็กสลับมิติ: dp[v] = น้ำหนักน้อยที่สุดที่ได้มูลค่า v แล้วหา v มากที่สุดที่ dp[v] <= W (AtCoder DP E) — การ "สลับว่าอะไรเป็น index อะไรเป็นค่า" เป็นทักษะสำคัญ

4. Longest Increasing Subsequence O(n log n)

// t[k] = ค่าท้ายที่เล็กที่สุดของ increasing subsequence ยาว k+1
vector<int> t;
for (int x : a) {
    auto it = lower_bound(t.begin(), t.end(), x);  // strictly increasing
    // non-decreasing: ใช้ upper_bound
    if (it == t.end()) t.push_back(x); else *it = x;
}
int lis = t.size();

หลายโจทย์แปลงเป็น LIS ได้: กล่องซ้อนกันสองมิติ (sort มิติแรก แล้วหา LIS มิติที่สอง — ถ้ามิติแรกเท่ากันให้ sort มิติที่สองจากมากไปน้อย), จำนวนลบน้อยสุดให้ array เรียง = n − LIS

5. Grid DP และ DP บนสองสาย

// Edit distance: dp[i][j] = จำนวน operation น้อยสุดเปลี่ยน a[0..i) เป็น b[0..j)
for (int i = 0; i <= n; i++) for (int j = 0; j <= m; j++) {
    if (i == 0 || j == 0) { dp[i][j] = i + j; continue; }
    dp[i][j] = min({ dp[i-1][j] + 1,                        // ลบ
                     dp[i][j-1] + 1,                        // เพิ่ม
                     dp[i-1][j-1] + (a[i-1] != b[j-1]) });  // แทน / ตรงกัน
}
// LCS: dp[i][j] = (a[i-1]==b[j-1]) ? dp[i-1][j-1]+1 : max(dp[i-1][j], dp[i][j-1])
// ต้องการตัวคำตอบจริง: ย้อนจาก dp[n][m] กลับไปดูว่ามาจาก transition ไหน

6. Interval DP

State dp[l][r] = คำตอบของช่วง [l, r] คำนวณเรียงตามความยาวช่วงจากสั้นไปยาว ใช้กับเกมเอาของจากสองฝั่ง, รวมกองของ (Slimes / matrix chain), ตัด/รวม palindrome — ปกติ O(n2) state × O(n) transition = O(n3) ใช้ได้เมื่อ n ≤ ~400

// Removal Game: dp[l][r] = (คะแนนเรา - คะแนนคู่แข่ง) ที่ดีที่สุดเมื่อเหลือ a[l..r] และเป็นตาเรา
for (int l = n - 1; l >= 0; l--)
    for (int r = l; r < n; r++)
        dp[l][r] = (l == r) ? a[l] : max(a[l] - dp[l+1][r], a[r] - dp[l][r-1]);
// คะแนนผู้เล่นแรก = (sum + dp[0][n-1]) / 2

// รวมกอง (Slimes): dp[l][r] = min over k of dp[l][k] + dp[k+1][r] + sum(l..r)
for (int len = 2; len <= n; len++)
    for (int l = 0; l + len - 1 < n; l++) {
        int r = l + len - 1; dp[l][r] = INF;
        for (int k = l; k < r; k++)
            dp[l][r] = min(dp[l][r], dp[l][k] + dp[k+1][r] + S(l, r));
    }

7. Bitmask DP

สัญญาณ: n ≤ 20 และต้องจำว่า "ใช้อะไรไปแล้วบ้าง" State = subset ที่ใช้ไปแล้ว (+ ตัวสุดท้ายถ้าลำดับมีผล)

// TSP / Hamiltonian path: dp[mask][v] = ต้นทุนน้อยสุดที่เยี่ยมชม mask แล้วจบที่ v
vector<vector<ll>> dp(1 << n, vector<ll>(n, INF));
dp[1][0] = 0;                                     // เริ่มที่ 0
for (int mask = 1; mask < (1 << n); mask++)
    for (int v = 0; v < n; v++) {
        if (!(mask >> v & 1) || dp[mask][v] == INF) continue;
        for (int u = 0; u < n; u++) if (!(mask >> u & 1))
            dp[mask | 1 << u][u] = min(dp[mask | 1 << u][u], dp[mask][v] + cost[v][u]);
    }
// O(2^n · n^2): n = 20 → ~4·10^8 ช้าไปหน่อย ใช้ array ธรรมดา + loop แน่นๆ จะทัน

รูปแบบอื่น: "แบ่งเป็นกลุ่มน้อยที่สุด" (Elevator Rides) ใช้ dp[mask] = {จำนวนรอบ, น้ำหนักรอบสุดท้าย} · broken profile (Counting Tilings) ใช้ mask แทนรอยต่อระหว่างคอลัมน์

8. Digit DP

สัญญาณ: "นับจำนวนเลขใน [L, R] ที่มีคุณสมบัติเกี่ยวกับหลัก" และ R ถึง 1018 — คำนวณ f(R) − f(L−1) โดย f(N) = จำนวนใน [0, N]

string s; ll memo[20][11][2][2]; bool seen[20][11][2][2];
// pos: หลักที่กำลังเลือก, prev: หลักก่อนหน้า (10 = ยังไม่มี), tight: ยังติดขอบ N อยู่ไหม, started: เริ่มเลขแล้วหรือยัง
ll go(int pos, int prev, bool tight, bool started) {
    if (pos == (int)s.size()) return 1;
    ll &res = memo[pos][prev][tight][started];
    if (seen[pos][prev][tight][started]) return res;
    seen[pos][prev][tight][started] = true; res = 0;
    int lim = tight ? s[pos] - '0' : 9;
    for (int d = 0; d <= lim; d++) {
        if (started && d == prev) continue;      // เงื่อนไขตัวอย่าง: ห้ามหลักติดกันซ้ำ (CSES Counting Numbers)
        bool ns = started || d != 0;
        res += go(pos + 1, ns ? d : 10, tight && d == lim, ns);
    }
    return res;
}
ll f(ll N) { if (N < 0) return 0; s = to_string(N); memset(seen, 0, sizeof seen); return go(0, 10, true, false); }
