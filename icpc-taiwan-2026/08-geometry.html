1. Point & cross product

struct P {
    ll x, y;
    P operator+(P o) const { return {x + o.x, y + o.y}; }
    P operator-(P o) const { return {x - o.x, y - o.y}; }
    P operator*(ll k) const { return {x * k, y * k}; }
    bool operator<(P o) const { return tie(x, y) < tie(o.x, o.y); }
    bool operator==(P o) const { return x == o.x && y == o.y; }
};
ll dot(P a, P b)   { return a.x * b.x + a.y * b.y; }
ll cross(P a, P b) { return a.x * b.y - a.y * b.x; }
// orientation ของ c เทียบกับเส้น a→b:  >0 ซ้าย (ทวนเข็ม), <0 ขวา, 0 อยู่บนเส้นเดียวกัน
ll orient(P a, P b, P c) { return cross(b - a, c - a); }
int sgn(ll v) { return (v > 0) - (v < 0); }
ทำไม cross product สำคัญที่สุด cross(a, b) = |a||b| sin θ → เครื่องหมายบอก "เลี้ยวซ้ายหรือขวา", ค่าคือสองเท่าของพื้นที่สามเหลี่ยม · dot(a, b) = |a||b| cos θ → เครื่องหมายบอก "มุมแหลม/ป้าน" · ทุกอย่างในหน้านี้สร้างจากสองฟังก์ชันนี้ และไม่ต้องใช้ sin/cos/sqrt เลย
ถ้าพิกัดถึง 109 cross product ถึง ~4·1018 — ยังพอดี long long แต่ถ้าคูณ cross สองตัวกัน ต้องใช้ __int128

2. Segment intersection

bool onSeg(P a, P b, P p) {        // p อยู่บน segment ab (รวมปลาย)
    return orient(a, b, p) == 0 &&
           min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}
bool segInter(P a, P b, P c, P d) {
    int d1 = sgn(orient(a, b, c)), d2 = sgn(orient(a, b, d));
    int d3 = sgn(orient(c, d, a)), d4 = sgn(orient(c, d, b));
    if (d1 * d2 < 0 && d3 * d4 < 0) return true;   // ตัดกันแบบข้ามจริง
    return onSeg(a, b, c) || onSeg(a, b, d) || onSeg(c, d, a) || onSeg(c, d, b);
}
// จุดตัดของเส้นตรง (ต้องใช้ทศนิยม): a + (b-a) * cross(c-a, d-c) / cross(b-a, d-c)

3. Polygon: area, point-in-polygon, Pick

// Shoelace: สองเท่าของพื้นที่ (เป็นจำนวนเต็มเสมอเมื่อจุดเป็นจำนวนเต็ม)
ll area2(const vector<P> &p) {
    ll s = 0; int n = p.size();
    for (int i = 0; i < n; i++) s += cross(p[i], p[(i + 1) % n]);
    return llabs(s);          // ถ้าไม่ abs: บวก = ทวนเข็ม, ลบ = ตามเข็ม
}

// Point in polygon (ray casting): 0 = นอก, 1 = ใน, 2 = บนขอบ
int inPoly(const vector<P> &p, P q) {
    int n = p.size(); bool in = false;
    for (int i = 0; i < n; i++) {
        P a = p[i], b = p[(i + 1) % n];
        if (onSeg(a, b, q)) return 2;
        if ((a.y > q.y) != (b.y > q.y)) {                 // ขอบคร่อมเส้นแนวนอนผ่าน q
            // เทียบตำแหน่งจุดตัดกับ q.x โดยไม่หาร: ใช้เครื่องหมายของ orient
            ll o = orient(a, b, q);
            if ((b.y > a.y) ? o > 0 : o < 0) in = !in;
        }
    }
    return in;
}

// Pick's theorem: A = I + B/2 - 1
// B (จุด lattice บนขอบ) = Σ gcd(|dx|, |dy|) ของทุกขอบ
// I (จุดภายใน) = (area2 - B + 2) / 2

4. Convex hull (Andrew's monotone chain)

vector<P> hull(vector<P> pts) {
    sort(pts.begin(), pts.end());
    pts.erase(unique(pts.begin(), pts.end()), pts.end());
    int n = pts.size();
    if (n < 3) return pts;
    vector<P> h(2 * n); int k = 0;
    for (int i = 0; i < n; i++) {                                   // lower hull
        while (k >= 2 && orient(h[k-2], h[k-1], pts[i]) <= 0) k--;  // <= 0: ตัดจุดที่อยู่บนเส้น
        h[k++] = pts[i];
    }
    for (int i = n - 2, t = k + 1; i >= 0; i--) {                  // upper hull
        while (k >= t && orient(h[k-2], h[k-1], pts[i]) <= 0) k--;
        h[k++] = pts[i];
    }
    h.resize(k - 1);                                                // ทวนเข็มนาฬิกา
    return h;
}
// ต้องการเก็บจุดที่อยู่บนขอบด้วย (CSES Convex Hull): เปลี่ยน <= 0 เป็น < 0
// (ระวังกรณีทุกจุดอยู่บนเส้นเดียวกัน — จุดจะถูกใส่ซ้ำ ต้องจัดการแยก)

ใช้ hull ต่อ: เส้นรอบรูปของรั้วที่ล้อมทุกจุด, diameter ของชุดจุด (rotating calipers), ตรวจว่าจุดอยู่ใน convex polygon ใน O(log n)

5. ทำงานกับทศนิยม

  • ใช้ long double และ const long double EPS = 1e-9; เทียบ a < b - EPS แทน a < b
  • เลี่ยง sqrt/atan2 ถ้าทำได้: เทียบระยะทางด้วยกำลังสอง, sort ตามมุมด้วย half-plane + cross
  • acos(x) เมื่อ x หลุดเป็น 1.0000000001 จะได้ NaN → clamp ก่อน
  • พิมพ์: cout << fixed << setprecision(9); ระวังพิมพ์ -0.000000
  • ถ้าพิกัดเป็นจำนวนเต็ม คำตอบที่เป็นพื้นที่มักเป็น "จำนวนเต็มหรือ .5" → ใช้ area2 แล้วพิมพ์เอง

6. เทคนิคอื่นที่ควรรู้จัก (ชื่อ + ไอเดีย)

  • Closest pair of points: sort ตาม x, sweep แล้วเก็บ set ของจุดในแถบกว้าง d เรียงตาม y → O(n log n)
  • Sweep ตามมุม: sort จุดรอบจุดศูนย์กลางด้วย comparator (แบ่งครึ่งระนาบแล้วใช้ cross)
  • วงกลม: จุดอยู่ในวงกลม ⇔ dist² ≤ r² (จำนวนเต็ม), จุดตัดวงกลม-เส้นต้องใช้ทศนิยม
  • Rotating calipers: ระยะห่างสูงสุดระหว่างจุดบน hull ใน O(n)
  • แหล่งอ้างอิง: KACTL geometry — template ที่ทดสอบแล้ว เหมาะใส่ notebook
