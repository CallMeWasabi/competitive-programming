deque<int> dq; d[s] = 0; dq.push_back(s);

3. Shortest paths

// Dijkstra
const ll INF = 1e18;
vector<vector<pair<int,ll>>> g(n);
vector<ll> dist(n, INF);
priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<>> pq;
dist[s] = 0; pq.push({0, s});
while (!pq.empty()) {
    auto [d, v] = pq.top(); pq.pop();

}

4. DSU & Minimum Spanning Tree

struct DSU {
    vector<int> p, sz;
    DSU(int n) : p(n), sz(n, 1) { iota(p.begin(), p.end(), 0); }
    int find(int x) { while (x != p[x]) x = p[x] = p[p[x]]; return x; }   // path halving
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a; sz[a] += sz[b];
        return true;
    }
};

// Kruskal
sort(edges.begin(), edges.end(), [](auto &x, auto &y) { return x.w < y.w; });
DSU dsu(n); ll cost = 0; int used = 0;
for (auto &e : edges) if (dsu.unite(e.u, e.v)) { cost += e.w; used++; }
if (used != n - 1) { /* กราฟไม่เชื่อมกัน → IMPOSSIBLE */ }
DSU ใช้ได้มากกว่า MSTประมวลผล query แบบ offline ย้อนเวลา (ลบ edge = เพิ่ม edge ถ้ามองถอยหลัง), เรียง query ตาม threshold ("ใช้ได้เฉพาะ edge น้ำหนัก ≤ x"), หรือเก็บข้อมูลรวมของกลุ่ม (ขนาด, min, parity สำหรับ bipartite แบบ online)

6. Strongly Connected Components & 2-SAT

// Tarjan: comp id ที่ได้เรียงแบบ reverse topological order (comp 0 คือ sink)
int timer = 0, nc = 0;
vector<int> tin(n, -1), low(n), comp(n, -1), stk;
vector<bool> on(n, false);
function<void(int)> dfs = [&](int v) {
    tin[v] = low[v] = timer++; stk.push_back(v); on[v] = true;
    for (int u : adj[v]) {
        if (tin[u] == -1) { dfs(u); low[v] = min(low[v], low[u]); }
        else if (on[u]) low[v] = min(low[v], tin[u]);
    }
    if (low[v] == tin[v]) {
        while (true) {
            int u = stk.back(); stk.pop_back(); on[u] = false; comp[u] = nc;
            if (u == v) break;
        }
        nc++;
    }
};
for (int v = 0; v < n; v++) if (tin[v] == -1) dfs(v);

2-SAT: ตัวแปร xi มี node สองตัว (2i = จริง, 2i+1 = เท็จ) เงื่อนไข (a ∨ b) → edge ¬a→b และ ¬b→a · ไม่มีคำตอบถ้า x กับ ¬x อยู่ SCC เดียวกัน · ถ้ามีคำตอบ ให้ x = จริง เมื่อ comp[2i] < comp[2i+1] (ใช้ลำดับของ Tarjan ด้านบน)

7. Bridges & articulation points

// adj[v] = {u, edge_id}: ใช้ edge id เพื่อรองรับ multi-edge
vector<int> tin(n, -1), low(n); int timer = 0;
vector<int> bridges; vector<bool> cut(n, false);
function<void(int,int)> dfs = [&](int v, int pe) {
    tin[v] = low[v] = timer++; int children = 0;
    for (auto [u, id] : adj[v]) {
        if (id == pe) continue;
        if (tin[u] != -1) { low[v] = min(low[v], tin[u]); continue; }
        dfs(u, id); children++;
        low[v] = min(low[v], low[u]);
        if (low[u] > tin[v]) bridges.push_back(id);                 // bridge
        if (pe != -1 && low[u] >= tin[v]) cut[v] = true;           // articulation point
    }
    if (pe == -1 && children > 1) cut[v] = true;                  // root
};

8. Bipartite matching & max flow

ระดับพื้นฐานต้องรู้จักโมเดลและมี template ที่เชื่อถือได้ใน notebook: คนกับงาน → matching · ตัด edge น้อยสุดให้ s ไม่ถึง t → min cut = max flow · จำนวน path ที่ไม่ใช้ edge ร่วมกัน → flow ความจุ 1 · node มีความจุ → แยก node เป็น in/out

struct Dinic {
    struct E { int to; ll cap; };
    vector<E> e; vector<vector<int>> g; vector<int> lvl, it;
    Dinic(int n) : g(n), lvl(n), it(n) {}
    void add(int u, int v, ll c) {                    // directed edge; ไม่มีทิศ: add(u,v,c), add(v,u,c)
        g[u].push_back(e.size()); e.push_back({v, c});
        g[v].push_back(e.size()); e.push_back({u, 0});
    }
    bool bfs(int s, int t) {
        fill(lvl.begin(), lvl.end(), -1); lvl[s] = 0;
        queue<int> q; q.push(s);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (int id : g[v]) if (e[id].cap > 0 && lvl[e[id].to] < 0) {
                lvl[e[id].to] = lvl[v] + 1; q.push(e[id].to);
            }
        }
        return lvl[t] >= 0;
    }
    ll dfs(int v, int t, ll f) {
        if (v == t) return f;
        for (int &i = it[v]; i < (int)g[v].size(); i++) {
            int id = g[v][i], u = e[id].to;
            if (e[id].cap > 0 && lvl[u] == lvl[v] + 1) {
                ll d = dfs(u, t, min(f, e[id].cap));
                if (d > 0) { e[id].cap -= d; e[id ^ 1].cap += d; return d; }
            }
        }
        return 0;
    }
    ll maxflow(int s, int t) {
        ll flow = 0;
        while (bfs(s, t)) {
            fill(it.begin(), it.end(), 0);
            while (ll f = dfs(s, t, LLONG_MAX)) flow += f;
        }
        return flow;
    }
};
// Min cut: หลัง maxflow, node ที่ lvl[v] != -1 (BFS ครั้งสุดท้ายไปถึง) คือฝั่ง s
// edge (u→v) ที่ u อยู่ฝั่ง s และ v อยู่ฝั่ง t คือ edge ที่ถูกตัด
// Bipartite matching: s→ซ้ายทุกตัว (cap 1), ซ้าย→ขวา (cap 1), ขวาทุกตัว→t (cap 1)
