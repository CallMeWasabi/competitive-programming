1. Prefix sum & difference array

p[i] = a[0] + ... + a[i-1] ทำให้ผลรวมช่วง [l, r) = p[r] - p[l] ใน O(1)

vector<ll> p(n + 1, 0);
for (int i = 0; i < n; i++) p[i + 1] = p[i] + a[i];
// sum(a[l..r]) รวมปลาย = p[r+1] - p[l]

// 2D: P[i][j] = ผลรวมสี่เหลี่ยม (0,0)..(i-1,j-1)
for (int i = 1; i <= n; i++) for (int j = 1; j <= m; j++)
    P[i][j] = g[i-1][j-1] + P[i-1][j] + P[i][j-1] - P[i-1][j-1];
// ผลรวม (r1,c1)..(r2,c2) รวมปลาย, 0-indexed:
// P[r2+1][c2+1] - P[r1][c2+1] - P[r2+1][c1] + P[r1][c1]

// difference array: บวก v ทั้งช่วง [l, r] หลายครั้ง แล้วค่อยถามทีเดียว
d[l] += v; d[r + 1] -= v;     // แล้ว prefix sum ของ d คือค่าจริง
Pattern ที่ต้องจำ: นับ subarray ที่ผลรวม = k subarray [l, r) มีผลรวม k ⇔ p[r] - p[l] = k ⇔ p[l] = p[r] - k วน r แล้วนับว่าเคยเห็น p[r]-k กี่ครั้งด้วย map — ใช้ได้กับค่าติดลบด้วย (ต่างจาก two pointers) เทคนิคเดียวกันใช้กับ "ผลรวมหาร n ลงตัว" (เก็บ p mod n) และ xor

2. Two pointers / sliding window

ใช้เมื่อ "ถ้าช่วง [l, r] ใช้ได้ ช่วงที่เล็กกว่าข้างในก็ใช้ได้" (monotone) — ขยับ r ไปเรื่อยๆ แล้วหด l จนกว่าจะกลับมาใช้ได้ รวม O(n)

// ช่วงยาวที่สุดที่ผลรวม <= S (ทุกค่า >= 0)
int best = 0; ll sum = 0;
for (int l = 0, r = 0; r < n; r++) {
    sum += a[r];
    while (sum > S) sum -= a[l++];
    best = max(best, r - l + 1);
}

// หาคู่ใน array ที่เรียงแล้วซึ่งรวมได้ x
int i = 0, j = n - 1;
while (i < j) {
    if (a[i] + a[j] == x) break;
    (a[i] + a[j] < x) ? i++ : j--;
}
Two pointers แบบผลรวมใช้ไม่ได้ถ้ามีค่าติดลบ — ใช้ prefix sum + map หรือ prefix sum + monotonic deque แทน

3. Binary search on answer

สัญญาณ: โจทย์ถาม "ค่าน้อยที่สุดที่ทำให้ ... ได้" หรือ "ค่ามากที่สุดของค่าต่ำสุด" และถ้าค่า t ใช้ได้ ค่าที่มากกว่า t ก็ใช้ได้ (monotone) — เปลี่ยนจาก หาคำตอบ เป็น ตรวจคำตอบ ซึ่งมักง่ายกว่ามาก (มักเป็น greedy O(n))

// หา t ที่น้อยที่สุดที่ ok(t) == true
// invariant: ok(lo) == false, ok(hi) == true  (เลือก lo/hi ให้จริงตาม invariant)
ll lo = 0, hi = 2e18;            // ระวัง: hi ต้องใหญ่พอที่จะ ok แน่ๆ
while (hi - lo > 1) {
    ll mid = lo + (hi - lo) / 2;
    if (ok(mid)) hi = mid; else lo = mid;
}
// คำตอบ = hi

// ตัวอย่าง ok() ของ Factory Machines: เวลา t ผลิตได้ครบ target ไหม
bool ok(ll t) {
    ll made = 0;
    for (ll k : machines) {
        made += t / k;
        if (made >= target) return true;   // ตัดก่อน overflow
    }
    return false;
}

// binary search บนจำนวนจริง: วนจำนวนรอบคงที่ ไม่ต้องใช้ epsilon
double L = 0, R = 1e9;
for (int it = 0; it < 100; it++) {
    double M = (L + R) / 2;
    (ok(M) ? R : L) = M;
}

แบบ invariant lo=false / hi=true แบบนี้ไม่มี off-by-one และไม่ติด loop ไม่รู้จบ — ใช้แบบเดียวตลอดชีวิต

4. Greedy & exchange argument

Greedy ผิดบ่อยที่สุดในทุกหัวข้อ เพราะ "ดูถูก" ตอนลองตัวอย่าง วิธีพิสูจน์ที่ใช้ได้เกือบทุกครั้งคือ exchange argument: สมมติคำตอบที่ดีที่สุดไม่ได้เลือกแบบ greedy แล้วแสดงว่าสลับให้เป็นแบบ greedy ได้โดยไม่แย่ลง

PatternGreedy ที่ถูก
เลือกช่วงที่ไม่ทับกันให้ได้มากที่สุดsort ตามจุดจบ เลือกทุกช่วงที่เริ่มหลังช่วงที่เลือกล่าสุด
งานมี deadline/เวลา ลด penalty รวมsort ตามเวลาทำ (shortest first) หรือหา key ด้วยการเทียบสลับสองงานติดกัน
จับคู่ของสองกลุ่ม (คนกับห้อง)sort ทั้งสองฝั่ง แล้ว two pointers / multiset lower_bound
เลือก k ตัวทีละขั้น ที่ตัวเลือกเปลี่ยนไปเรื่อยๆpriority_queue
เหรียญ/ยอดรวมที่ทำได้sort แล้วตรวจว่า a[i] <= sum+1 (Missing Coin Sum)
เทคนิคหา key ของการ sort พิจารณางานสองงาน A, B ที่อยู่ติดกัน เขียนค่า cost ของลำดับ AB และ BA แล้วหาเงื่อนไขที่ AB ≤ BA เช่น ได้ a.t * b.w < b.t * a.w → นั่นคือ comparator (ต้องเช็คว่าเป็น strict weak ordering)
ถ้าคิด greedy ได้แต่พิสูจน์ไม่ได้ ให้เขียน brute force ขนาดเล็กแล้ว stress test ก่อน submit — 5 นาทีนี้ถูกกว่า penalty 20 นาที

5. Sweep line

เปลี่ยนช่วง [l, r] เป็น event สองตัว (+1 ที่ l, −1 ที่ r) แล้ว sort ตามเวลา เดินจากซ้ายไปขวา — ใช้นับ "จุดที่ทับกันมากที่สุด", จัดห้อง, union ความยาวช่วง

vector<pair<int,int>> ev;
for (auto [l, r] : seg) { ev.push_back({l, +1}); ev.push_back({r, -1}); }
sort(ev.begin(), ev.end());     // ที่เวลาเท่ากัน -1 มาก่อน +1 (ถ้าจุดปลายไม่นับว่าทับ)
int cur = 0, best = 0;
for (auto [t, d] : ev) best = max(best, cur += d);
