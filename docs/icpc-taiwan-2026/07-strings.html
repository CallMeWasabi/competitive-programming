1. Polynomial hashing

hash(s) = s0Bn−1 + s1Bn−2 + … + sn−1 (mod M) · เก็บ prefix hash แล้ว hash ของ substring ใดก็ได้คำนวณใน O(1) · ใช้ M = 261−1 และ B สุ่ม เพื่อไม่ให้โดน anti-hash test และโอกาสชนต่ำมาก

using ull = unsigned long long;
const ull HM = (1ULL << 61) - 1;
ull hmul(ull a, ull b) {
    __uint128_t c = (__uint128_t)a * b;
    ull r = (ull)(c & HM) + (ull)(c >> 61);
    return r >= HM ? r - HM : r;
}
const ull HB = mt19937_64(chrono::steady_clock::now().time_since_epoch().count())() % (HM - 1000) + 500;

struct Hash {
    vector<ull> h, pw;
    Hash(const string &s) : h(s.size() + 1, 0), pw(s.size() + 1, 1) {
        for (size_t i = 0; i < s.size(); i++) {
            h[i + 1] = hmul(h[i], HB) + (unsigned char)s[i];
            if (h[i + 1] >= HM) h[i + 1] -= HM;
            pw[i + 1] = hmul(pw[i], HB);
        }
    }
    ull get(int l, int r) {          // hash ของ s[l, r)
        ull x = h[r] + HM - hmul(h[l], pw[r - l]);
        return x >= HM ? x - HM : x;
    }
};
สิ่งที่ hashing ทำได้
  • เทียบ substring สองตัวว่าเท่ากันไหม O(1)
  • LCP ของสอง suffix: binary search ความยาว O(log n) → เปรียบเทียบ lexicographic ของ substring ได้ใน O(log n)
  • palindrome check: hash ของ s เทียบกับ hash ของ reverse(s) ในช่วงที่สมมาตร
  • "substring ยาว k ที่ซ้ำกันมีไหม" → ใส่ hash ทุกตัวใน set/sort, แล้ว binary search บน k
  • hash ของ multiset/set: ผลรวมของค่าสุ่มที่ map ให้แต่ละสมาชิก
อย่าใช้ M = 264 (ปล่อย overflow) — มี test แบบ Thue–Morse ที่ทำให้ชนแน่นอน และอย่าใช้ base คงที่ที่คนอื่นเดาได้ (สำคัญน้อยใน ICPC แต่เป็นนิสัยที่ดี)

2. Prefix function (KMP)

pi[i] = ความยาว prefix ที่ยาวที่สุดของ s[0..i] ที่เป็น suffix ด้วย (ไม่นับตัวเอง)

vector<int> prefix_function(const string &s) {
    int n = s.size(); vector<int> pi(n, 0);
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j > 0 && s[i] != s[j]) j = pi[j - 1];
        if (s[i] == s[j]) j++;
        pi[i] = j;
    }
    return pi;
}
// ค้นหา pattern p ใน text t: คำนวณบน p + '#' + t
// ตำแหน่ง i ที่ pi[i] == |p| คือจุดจบของ occurrence
// Borders ทั้งหมดของ s: เริ่ม k = pi[n-1] แล้ว k = pi[k-1] ซ้ำจน 0
// คาบที่สั้นที่สุด: n - pi[n-1] (เป็นคาบสมบูรณ์ถ้าหาร n ลงตัว)

3. Z-function

z[i] = ความยาว prefix ร่วมที่ยาวที่สุดระหว่าง s กับ s[i..]

vector<int> z_function(const string &s) {
    int n = s.size(); vector<int> z(n, 0);
    for (int i = 1, l = 0, r = 0; i < n; i++) {
        if (i < r) z[i] = min(r - i, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
        if (i + z[i] > r) l = i, r = i + z[i];
    }
    return z;   // z[0] = 0 ตามธรรมเนียม (จริงๆ คือ n)
}
// คาบ: i เป็นคาบ ⇔ i + z[i] == n

4. Trie

struct Trie {
    vector<array<int, 26>> nx; vector<int> endc;
    Trie() { nx.push_back({}); nx[0].fill(-1); endc.push_back(0); }
    void insert(const string &s) {
        int v = 0;
        for (char ch : s) {
            int c = ch - 'a';
            if (nx[v][c] == -1) {
                nx[v][c] = nx.size();
                nx.push_back({}); nx.back().fill(-1); endc.push_back(0);
            }
            v = nx[v][c];
        }
        endc[v]++;
    }
};
// Word Combinations: dp[i] = จำนวนวิธีแบ่ง s[i..]; เดิน trie จาก i ไปทางขวา ทุก node ที่ endc > 0 ที่ตำแหน่ง j → dp[i] += dp[j+1]
// Binary trie (bit 30→0): หา max XOR ของคู่ใน O(n · 30)

5. Manacher (palindrome ทุกจุดศูนย์กลาง O(n))

// แทรก '#' เพื่อรวมกรณีความยาวคู่/คี่
string t = "#";
for (char c : s) { t += c; t += '#'; }
int m = t.size(); vector<int> p(m, 0);
for (int i = 0, l = 0, r = -1; i < m; i++) {
    int k = (i > r) ? 1 : min(p[l + r - i], r - i + 1);
    while (i - k >= 0 && i + k < m && t[i - k] == t[i + k]) k++;
    p[i] = k--;
    if (i + k > r) l = i - k, r = i + k;
}
// palindrome ใน s ที่ศูนย์กลาง i ของ t ยาว p[i] - 1
// เริ่มที่ index (i - (p[i] - 1)) / 2 ใน s

ถ้าจำ Manacher ไม่ได้ในวันแข่ง: hash + binary search รัศมีแต่ละศูนย์กลาง = O(n log n) ก็มักพอ

6. ไปต่อ

  • Suffix array + LCP — นับ substring ที่ต่างกัน, substring ซ้ำที่ยาวที่สุด (repo มี challenging_prog/week_10/dvaput.cpp)
  • Aho–Corasick — ค้นหาหลาย pattern พร้อมกัน, DP บน automaton
  • KMP automaton + DP — นับสตริงยาว n ที่ (ไม่)มี pattern (Required Substring)

7. แบบฝึกหัด

  • String Matching
    KMP หรือ Z หรือ hash
  • Finding Borders
    prefix function
  • Finding Periods
    Z-function
  • Word Combinations
    trie + DP
  • Minimal Rotation
    hash + LCP compare หรือ Booth
  • Longest Palindrome
    Manacher
  • Required Substring
    KMP automaton + DP
  • Palindrome Queries
    hash + BIT
  • Distinct Substrings
    suffix array
  • Password
    prefix function / Z
  • Compress Words
    KMP หรือ hash ระวังเวลา
  • Good Substrings
    trie หรือ hash
