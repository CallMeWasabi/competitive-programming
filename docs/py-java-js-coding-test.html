Which language to pick

Pick Python as your main language and keep Java as a backup for loop-heavy problems. Three questions with partly visible test cases usually reward writing correct code fast over raw speed, and Python is the shortest path from a C++ idea to working code.

Rules of thumb:

  • Default to Python. Typical test limits (n up to ~10^5 with O(n log n)) pass comfortably.

  • Switch to Java if you need an ordered set/map with lower_bound, or O(n^2) with n around 5,000+, or deep recursion.

  • Avoid JavaScript unless you already know it well: no heap, no ordered map, and integer precision traps.

The test may be function-style (you fill in def solve(arr) and return a value, like LeetCode) or stdin/stdout (like Codeforces). Read the template first; with a function template, do not print or read input yourself.

Program skeleton and fast I/O

Each block reads n, then n integers, and prints their sum. Memorize these four; everything else plugs into them.

Java trap: Arrays.sort(int[]) is a dual-pivot quicksort with O(n^2) worst case on adversarial tests. If you fear hacked inputs, sort an Integer[] or long[] after shuffling. Use Integer.compare(a, b), never a - b, when values can overflow.

lower_bound you can paste in Java / JavaScript (first index with a[i] >= x):

static int lowerBound(int[] a, int x) {
    int lo = 0, hi = a.length;
    while (lo < hi) { int mid = (lo + hi) >>> 1; if (a[mid] < x) lo = mid + 1; else hi = mid; }
    return lo;
}
function lowerBound(a, x) {
  let lo = 0, hi = a.length;
  while (lo < hi) { const mid = (lo + hi) >> 1; if (a[mid] < x) lo = mid + 1; else hi = mid; }
  return lo;
}

For upper_bound, change a[mid] < x to a[mid] <= x. Binary search on the answer works the same in all four languages.

Algorithm templates

The logic is identical to your C++ templates; only the containers change. Graphs below are 0-indexed with n nodes.

BFS (shortest path, unweighted)

from collections import deque
def bfs(adj, s):
    dist = [-1] * len(adj); dist[s] = 0
    q = deque([s])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1; q.append(v)
    return dist
static int[] bfs(List<Integer>[] adj, int s) {
    int[] dist = new int[adj.length]; Arrays.fill(dist, -1); dist[s] = 0;
    ArrayDeque<Integer> q = new ArrayDeque<>(); q.add(s);
    while (!q.isEmpty()) {
        int u = q.poll();
        for (int v : adj[u]) if (dist[v] == -1) { dist[v] = dist[u] + 1; q.add(v); }
    }
    return dist;
}
// build: List<Integer>[] adj = new ArrayList[n]; for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
function bfs(adj, s) {
  const dist = new Array(adj.length).fill(-1); dist[s] = 0;
  const q = [s]; let head = 0;            // never q.shift(): O(n)
  while (head < q.length) {
    const u = q[head++];
    for (const v of adj[u]) if (dist[v] === -1) { dist[v] = dist[u] + 1; q.push(v); }
  }
  return dist;
}
// build: const adj = Array.from({length: n}, () => []);

Dijkstra

import heapq
def dijkstra(adj, s):                     # adj[u] = [(v, w), ...]
    dist = [float('inf')] * len(adj); dist[s] = 0
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue
        for v, w in adj[u]:
            if d + w < dist[v]:
                dist[v] = d + w; heapq.heappush(pq, (dist[v], v))
    return dist
static long[] dijkstra(List<int[]>[] adj, int s) {   // adj[u] holds {v, w}
    long[] dist = new long[adj.length]; Arrays.fill(dist, Long.MAX_VALUE); dist[s] = 0;
    PriorityQueue<long[]> pq = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
    pq.add(new long[]{0, s});
    while (!pq.isEmpty()) {
        long[] top = pq.poll(); long d = top[0]; int u = (int) top[1];
        if (d > dist[u]) continue;
        for (int[] e : adj[u]) if (d + e[1] < dist[e[0]]) {
            dist[e[0]] = d + e[1]; pq.add(new long[]{dist[e[0]], e[0]});
        }
    }
    return dist;
}
// JavaScript has no priority queue: paste this MinHeap (O(log n) push/pop)
class MinHeap {
  constructor(cmp = (a, b) => a - b) { this.h = []; this.cmp = cmp; }
  size() { return this.h.length; }
  peek() { return this.h[0]; }
  push(x) {
    const h = this.h; h.push(x); let i = h.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.cmp(h[i], h[p]) >= 0) break;
      [h[i], h[p]] = [h[p], h[i]]; i = p;
    }
  }
  pop() {
    const h = this.h, top = h[0], last = h.pop();
    if (h.length) {
      h[0] = last; let i = 0;
      for (;;) {
        const l = 2 * i + 1, r = l + 1; let m = i;
        if (l < h.length && this.cmp(h[l], h[m]) < 0) m = l;
        if (r < h.length && this.cmp(h[r], h[m]) < 0) m = r;
        if (m === i) break;
        [h[i], h[m]] = [h[m], h[i]]; i = m;
      }
    }
    return top;
  }
}

function dijkstra(adj, s) {               // adj[u] = [[v, w], ...]
  const dist = new Array(adj.length).fill(Infinity); dist[s] = 0;
  const pq = new MinHeap((a, b) => a[0] - b[0]); pq.push([0, s]);
  while (pq.size()) {
    const [d, u] = pq.pop();
    if (d > dist[u]) continue;
    for (const [v, w] of adj[u]) if (d + w < dist[v]) { dist[v] = d + w; pq.push([dist[v], v]); }
  }
  return dist;
}

Disjoint set union

parent = list(range(n)); size = [1] * n
def find(x):
    while parent[x] != x:
        parent[x] = parent[parent[x]]     # path halving, no recursion
        x = parent[x]
    return x
def union(a, b):
    a, b = find(a), find(b)
    if a == b: return False
    if size[a] < size[b]: a, b = b, a
    parent[b] = a; size[a] += size[b]; return True
static int[] parent, sz;
static int find(int x) { while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; } return x; }
static boolean union(int a, int b) {
    a = find(a); b = find(b); if (a == b) return false;
    if (sz[a] < sz[b]) { int t = a; a = b; b = t; }
    parent[b] = a; sz[a] += sz[b]; return true;
}

function union(a, b) {
  a = find(a); b = find(b); if (a === b) return false;
  if (sz[a] < sz[b]) [a, b] = [b, a];
  parent[b] = a; sz[a] += sz[b]; return true;
}

Gotchas and performance traps

Budget roughly 10^7 simple operations per second in Python, ~10^8 in Java and JavaScript, versus ~10^8–10^9 in C++.

Python

  • [[0] * m] * n makes n references to the same row. Use a list comprehension.

  • Default arguments are shared: never def f(a=[]).

  • input() without sys.stdin.readline is slow; readline keeps the \n, so call .strip() on strings.

  • Recursion over ~1,000 deep raises RecursionError; setrecursionlimit helps but can segfault past ~10^5. Go iterative.

  • Speed: local variables beat globals (wrap in main()); built-ins (sum, sorted, min, slicing, str.join) run in C; avoid list.pop(0) and x in list (both O(n)), use deque and set.

  • round(2.5) = 2 (banker's rounding). Use int(x + 0.5) for positives.

  • Python 3.8+ has math.comb, math.perm; 3.10+ has int.bit_count; 3.9+ has functools.cache. Check the version if the platform shows it.

Java

  • == on String, Integer, Long compares references. Use .equals().

  • Generic collections hold boxed objects: ArrayList<Integer> is 3–5x slower than int[]. Use primitive arrays in hot loops.

  • Integer division by zero throws; int overflow wraps silently.

  • Always out.flush() (or out.close()) at the end of a PrintWriter.

  • System.out.println in a loop of 10^5+ lines is slow: use PrintWriter or a StringBuilder.

  • Function-style platforms want the exact class and method name; don't add public to a second top-level class in the same file.

JavaScript

  • arr.sort() without a comparator sorts as strings.

  • == coerces types ("1" == 1 is true). Always use ===.

  • Numbers are doubles: products over 2^53 (≈ 9×10^15) silently lose precision; % MOD on them is wrong. Use BigInt for the multiply step.

  • Math.max(...arr) throws "Maximum call stack" on large arrays. Use a loop or reduce.

  • array.shift() and array.unshift() are O(n); array.includes() is O(n); use Set.

  • Object keys are strings: obj[1] and obj["1"] are the same key; Map keeps types.

  • for (const i in arr) gives string indices. Use for (let i = 0; ...) or for (const x of arr).

  • Typed arrays (Int32Array, Float64Array) are much faster for large numeric tables.

Exam-day checklist

Only part of the test cases are visible, so the hidden ones decide your score: test edge cases yourself before submitting.

Before the test

  • Solve 5–10 easy/medium problems in Python (LeetCode or HackerRank) to build muscle memory for dict, heapq, bisect, slicing.

  • Solve 2–3 in Java to rehearse the skeleton, TreeMap, PriorityQueue, and long arithmetic.

  • Retype the templates in this doc once from memory.

For each question

  • Read the code template: function-style (return the value) or stdin/stdout (read and print)? Which class/function name is required?

  • Note the limits and choose complexity as in C++, then apply the Python budget (~10^7 ops/s).

  • Run the visible test cases first.

  • Add your own hidden-case checks: n = 0 or 1, all equal values, negatives, maximum n, maximum values (overflow in Java/JS), empty string, duplicates.

  • Python: negative // and %? Java: int overflow, == on objects, flush()? JavaScript: default sort(), numbers over 2^53?

  • Remove debug prints before submitting; output format exactly as asked (spaces, newlines, decimals).

  • Partial credit counts: submit a correct brute force early, then optimize.
